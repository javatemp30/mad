package com.example.mad_2;

import androidx.appcompat.app.AppCompatActivity;
import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import java.util.regex.Pattern;

public class MainActivity extends AppCompatActivity {

    EditText res;
    String operator = "";
    boolean operatorPressed = false;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        res = findViewById(R.id.res);

        // Number buttons
        findViewById(R.id.btn0).setOnClickListener(numberListener);
        findViewById(R.id.btn1).setOnClickListener(numberListener);
        findViewById(R.id.btn2).setOnClickListener(numberListener);
        findViewById(R.id.btn3).setOnClickListener(numberListener);
        findViewById(R.id.btn4).setOnClickListener(numberListener);
        findViewById(R.id.btn5).setOnClickListener(numberListener);
        findViewById(R.id.btn6).setOnClickListener(numberListener);
        findViewById(R.id.btn7).setOnClickListener(numberListener);
        findViewById(R.id.btn8).setOnClickListener(numberListener);
        findViewById(R.id.btn9).setOnClickListener(numberListener);
        findViewById(R.id.dot).setOnClickListener(numberListener);

        // Operators
        findViewById(R.id.add).setOnClickListener(operatorListener);
        findViewById(R.id.minus).setOnClickListener(operatorListener);
        findViewById(R.id.mult).setOnClickListener(operatorListener);
        findViewById(R.id.div).setOnClickListener(operatorListener);

        // Equal
        findViewById(R.id.equ).setOnClickListener(equalListener);

        findViewById(R.id.ac).setOnClickListener(v -> {
            res.setText("");
            operator = "";
            operatorPressed = false;
        });
    }

    View.OnClickListener numberListener = v -> {
        Button b = (Button) v;
        String current = res.getText().toString();
        res.setText(current + b.getText().toString());
    };


    View.OnClickListener operatorListener = v -> {
        Button b = (Button) v;
        String current = res.getText().toString();

        if (current.isEmpty()) return;
        if (!operatorPressed) {
            operator = b.getText().toString();
            res.setText(current + operator);
            operatorPressed = true;
        }
    };

    // Equal Listener (=)
    View.OnClickListener equalListener = v -> {
        try {
            String expression = res.getText().toString(); // e.g., 8.2+5

            if (!operatorPressed) return;

            // Split safely EVEN IF numbers have dots
            String[] parts = expression.split(Pattern.quote(operator));

            if (parts.length < 2) {
                res.setText("Error");
                return;
            }

            double num1 = Double.parseDouble(parts[0]);
            double num2 = Double.parseDouble(parts[1]);

            double result = 0;

            switch (operator) {
                case "+":
                    result = num1 + num2;
                    break;

                case "-":
                    result = num1 - num2;
                    break;

                case "X":
                    result = num1 * num2;
                    break;

                case "/":
                    if (num2 == 0) {
                        res.setText("Error");
                        return;
                    }
                    result = num1 / num2;
                    break;
            }

            // Show result
            res.setText(String.valueOf(result));

            // Reset after calculation
            operatorPressed = false;
            operator = "";

        } catch (Exception e) {
            res.setText("Error");
        }
    };
}
